{{- range .Values.ingress.hosts }}
{{- $host := .host }}
{{- range .paths }}
     {{ $scheme }}://{{ $host }}{{ .path }}
{{- end }}

{{- else }}

     kubectl --namespace {{ .Release.Namespace }} port-forward svc/{{ include "dops-assistant.fullname" . }} {{ .Values.service.httpPort }}:{{ .Values.service.httpPort }}

   Then open http://localhost:{{ .Values.service.httpPort }}

{{- end }}

3. Health check:

     kubectl --namespace {{ .Release.Namespace }} exec deploy/{{ include "dops-assistant.fullname" . }} -- \
       wget -q -O - http://localhost:{{ .Values.service.httpPort }}/api/health
